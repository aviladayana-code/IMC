  (function initAnthropometricCalculator() {
    const weightInput = document.getElementById('input-weight');
    const heightInput = document.getElementById('input-height');
    const ageInput = document.getElementById('input-age');
    const metersDisplay = document.getElementById('display-meters');
    const calculatedBmi = document.getElementById('calculated-bmi');
    const mathTrace = document.getElementById('math-trace');
    const bmiBadge = document.getElementById('bmi-status-badge');
    const bmiText = document.getElementById('bmi-status-text');
    const pediatricAlert = document.getElementById('alert-pediatric');
    const dateInput = document.getElementById('input-date');

    // Default current date
    if (dateInput && !dateInput.value) {
      dateInput.value = new Date().toISOString().split('T')[0];
    }

    function recalculate() {
      const weight = parseFloat(weightInput.value) || 0;
      const heightCm = parseFloat(heightInput.value) || 0;
      const age = parseInt(ageInput.value, 10) || 0;

      // Pediatric visibility
      if (pediatricAlert) {
        if (age > 0 && age < 20) {
          pediatricAlert.classList.remove('hidden');
        } else {
          pediatricAlert.classList.add('hidden');
        }
      }

      // Height in meters
      const meters = heightCm > 0 ? (heightCm / 100) : 0;
      if (metersDisplay) {
        metersDisplay.textContent = 'Equivale a: ' + meters.toFixed(3) + ' m';
      }

      // BMI calculation
      if (weight > 0 && meters > 0) {
        const bmi = weight / Math.pow(meters, 2);
        const bmiFormatted = bmi.toFixed(2);
        
        if (calculatedBmi) calculatedBmi.textContent = bmiFormatted;
        if (mathTrace) mathTrace.textContent = weight.toFixed(1) + ' / (' + meters.toFixed(3) + ')²';

        // Categorization OMS
        let category = '';
        let dotColorClass = 'bg-secondary';

        if (bmi < 18.5) {
          category = 'Bajo peso (< 18.5 kg/m²)';
          dotColorClass = 'bg-secondary-fixed';
        } else if (bmi >= 18.5 && bmi < 25.0) {
          category = 'Normopeso (18.5 - 24.9 kg/m²)';
          dotColorClass = 'bg-on-tertiary-container';
        } else if (bmi >= 25.0 && bmi < 30.0) {
          category = 'Sobrepeso grado I (25.0 - 29.9 kg/m²)';
          dotColorClass = 'bg-secondary-container';
        } else if (bmi >= 30.0 && bmi < 35.0) {
          category = 'Obesidad Clase I (30.0 - 34.9 kg/m²)';
          dotColorClass = 'bg-surface-tint';
        } else {
          category = 'Obesidad Severa (≥ 35.0 kg/m²)';
          dotColorClass = 'bg-error';
        }

        if (bmiText) bmiText.textContent = category;
        const dot = bmiBadge.querySelector('span:first-child');
        if (dot) {
          dot.className = 'w-2.5 h-2.5 rounded-full ' + dotColorClass;
        }
      } else {
        if (calculatedBmi) calculatedBmi.textContent = '--';
        if (mathTrace) mathTrace.textContent = '-- / --';
        if (bmiText) bmiText.textContent = 'Ingrese peso y talla válidos';
      }
    }

    // Steppers listeners
    document.querySelectorAll('.btn-step-weight').forEach(btn => {
      btn.addEventListener('click', () => {
        const delta = parseFloat(btn.getAttribute('data-delta'));
        const current = parseFloat(weightInput.value) || 70;
        weightInput.value = Math.max(20, Math.min(300, current + delta)).toFixed(1);
        recalculate();
      });
    });

    document.querySelectorAll('.btn-step-height').forEach(btn => {
      btn.addEventListener('click', () => {
        const delta = parseFloat(btn.getAttribute('data-delta'));
        const current = parseFloat(heightInput.value) || 170;
        heightInput.value = Math.max(50, Math.min(250, current + delta)).toFixed(1);
        recalculate();
      });
    });

    // Inputs triggers
    weightInput.addEventListener('input', recalculate);
    heightInput.addEventListener('input', recalculate);
    ageInput.addEventListener('input', recalculate);

    // Biological sex selector toggles
    const sexM = document.getElementById('sex-m');
    const sexF = document.getElementById('sex-f');

    function selectSex(isMale) {
      if (isMale) {
        sexM.className = 'btn-sex py-space-sm px-space-md rounded-lg flex items-center justify-center gap-space-xs font-label-md text-label-md transition-colors bg-primary text-on-primary shadow-sm';
        sexF.className = 'btn-sex py-space-sm px-space-md rounded-lg flex items-center justify-center gap-space-xs font-label-md text-label-md transition-colors bg-surface-container text-on-surface-variant shadow-sm';
      } else {
        sexF.className = 'btn-sex py-space-sm px-space-md rounded-lg flex items-center justify-center gap-space-xs font-label-md text-label-md transition-colors bg-primary text-on-primary shadow-sm';
        sexM.className = 'btn-sex py-space-sm px-space-md rounded-lg flex items-center justify-center gap-space-xs font-label-md text-label-md transition-colors bg-surface-container text-on-surface-variant shadow-sm';
      }
    }

    if (sexM && sexF) {
      sexM.addEventListener('click', () => selectSex(true));
      sexF.addEventListener('click', () => selectSex(false));
    }

    // Context tags toggles
    document.querySelectorAll('.btn-check-tag').forEach(tag => {
      tag.addEventListener('click', () => {
        const isActive = tag.getAttribute('data-active') === 'true';
        if (isActive) {
          tag.setAttribute('data-active', 'false');
          tag.className = 'btn-check-tag px-3 py-1.5 rounded-full font-label-sm text-label-sm bg-surface-container text-on-surface-variant hover:bg-surface-container-high flex items-center gap-1';
        } else {
          tag.setAttribute('data-active', 'true');
          tag.className = 'btn-check-tag px-3 py-1.5 rounded-full font-label-sm text-label-sm bg-primary text-on-primary shadow-sm flex items-center gap-1';
        }
      });
    });

    // Toggle New / Search Subject
    const btnToggle = document.getElementById('btn-toggle-new-user');
    const profileCard = document.getElementById('profile-card');
    const userText = document.getElementById('btn-user-text');
    let isCreatingNew = false;

    if (btnToggle && profileCard) {
      btnToggle.addEventListener('click', () => {
        isCreatingNew = !isCreatingNew;
        if (isCreatingNew) {
          userText.textContent = 'Buscar Existente';
          profileCard.innerHTML = `
            <div class="flex items-center gap-space-sm w-full py-1">
              <span class="material-symbols-outlined text-secondary text-[24px]">person_add</span>
              <input type="text" placeholder="Nombre completo del nuevo sujeto..." class="w-full bg-transparent font-label-lg text-label-lg text-on-surface outline-none placeholder:text-outline-variant font-medium">
            </div>
          `;
        } else {
          userText.textContent = 'Crear Nuevo';
          profileCard.innerHTML = `
            <div class="flex items-center gap-space-sm min-w-0">
              <div class="w-11 h-11 rounded-full bg-surface-variant text-on-surface flex items-center justify-center font-headline-sm text-headline-sm flex-shrink-0">
                CM
              </div>
              <div class="min-w-0">
                <div class="flex items-center gap-space-xs">
                  <span class="font-label-lg text-label-lg text-on-surface truncate font-semibold">Carlos Mendoza</span>
                  <span class="bg-surface-container-highest text-on-surface-variant font-label-sm text-label-sm px-1.5 py-0.5 rounded">AT-2041</span>
                </div>
                <p class="font-body-sm text-body-sm text-on-surface-variant truncate">Rugby Union · Primera División</p>
              </div>
            </div>
            <button type="button" class="w-8 h-8 rounded-full bg-surface-container-lowest flex items-center justify-center text-on-surface-variant hover:text-secondary shadow-sm ml-2 flex-shrink-0">
              <span class="material-symbols-outlined text-[18px]">sync_alt</span>
            </button>
          `;
        }
      });
    }

    // Submit micro-feedback
    const submitBtn = document.getElementById('btn-submit-eval');
    if (submitBtn) {
      submitBtn.addEventListener('click', () => {
        const originalContent = submitBtn.innerHTML;
        submitBtn.innerHTML = `
          <span class="material-symbols-outlined animate-spin text-[22px]">progress_activity</span>
          <span>Procesando Diagnóstico...</span>
        `;
        submitBtn.classList.add('opacity-90');

        setTimeout(() => {
          submitBtn.innerHTML = `
            <span class="material-symbols-outlined text-on-tertiary-container text-[22px]">check_circle</span>
            <span>¡Evaluación Registrada con Éxito!</span>
          `;
          submitBtn.classList.remove('bg-primary');
          submitBtn.classList.add('bg-tertiary-container');
        }, 650);
      });
    }

    // Initial calculation call
    recalculate();
  })();
